export type ProjectRole = 'admin' | 'owner' | 'editor' | 'viewer';

export interface AuthResult {
  project: Project;
  user: UserInfo;
  role: ProjectRole;

  isAdmin: boolean;
}

export async function getAuthorizedProject(projectId: string): Promise<AuthResult>;

Viewer guard pattern to apply in each mutation action:
```typescript
// EXISTING pattern in most server actions:
try { await getAuthorizedProject(projectId); } catch { return { success: false, error: "Project not found" }; }

// NEW pattern — change to:
const auth = await getAuthorizedProject(projectId);
// auth is returned (notFound() already thrown if no access)
// ... but we need try-catch for the notFound() throw:
let auth: AuthResult;
try { auth = await getAuthorizedProject(projectId); } catch { return { success: false, error: "Project not found" }; }
if (!auth.canEdit) { return { success: false, error: "Read-only access" }; }
```

  <action>
    For each file, identify every **mutation function** (listed below) and add a viewer guard. The pattern depends on how the function currently calls `getAuthorizedProject`.

    **Pattern A — Functions using try-catch around getAuthorizedProject:**
    Most mutation functions currently have:
    ```typescript
    try { await getAuthorizedProject(projectId); } catch { return { success: false, error: "Project not found" }; }
    ```
    Change to:
    ```typescript
    let auth;
    try { auth = await getAuthorizedProject(projectId); } catch { return { success: false, error: "Project not found" }; }
    if (!auth.canEdit) { return { success: false, error: "Read-only access" }; }
    ```

    **Pattern B — Functions using bare await (no try-catch):**
    Some functions call `await getAuthorizedProject(projectId)` without try-catch (letting notFound() propagate).
    Change to:
    ```typescript
    const { canEdit } = await getAuthorizedProject(projectId);
    if (!canEdit) { return { success: false, error: "Read-only access" }; }
    ```

    **Specific mutations to guard per file:**

    **server/actions/uploads.ts** — Guard: `createUploadFromText`, `createUploadFromCSV`, `deleteUpload`. Do NOT guard `getUpload` (read-only).

    **server/actions/analysis.ts** — Guard: `analyzeProject`, `retryRun`, `cancelRun`. Do NOT guard `getRunProgress`, `getActiveRunForProject`, `getRunHistory` (read-only). Note: `retryRun` and `cancelRun` resolve the run's projectId from the run record before calling `getAuthorizedProject`.

    **server/actions/generation.ts** — Guard: `generateEpicsForProject`, `generateStoriesForEpic`. Do NOT guard `getRun`, `getRunsForProject` (read-only). Note: `generateStoriesForEpic` looks up the epic first to get `epic.projectId`.

    **server/actions/batch-stories.ts** — Guard: `startBatchStoryGeneration`, `retryBatchStoryRun`, `cancelBatchStoryRun`. Do NOT guard `getBatchStoryProgress`, `getActiveBatchStoryRun`, `getBatchStoryRunHistory` (read-only).

    **server/actions/subtasks.ts** — Guard: `generateSubtasksForEpic`, `retryBatchSubtaskRun`, `cancelBatchSubtaskRun`. Do NOT guard `getBatchSubtaskProgress`, `getActiveSubtaskRun`, `getBatchSubtaskRunHistory` (read-only).

    **server/actions/projects.ts** — Guard: `updateProject`, `deleteProject`. Do NOT guard `createProject` (uses getCurrentUser, no existing project to authorize against — per pitfall 5), `getProjects`, `getProjectName`, `getProject` (read-only).

    **Important:** Do NOT change the import statement if `getAuthorizedProject` is already imported. If the file only imports it by name, no change needed. The return type is extended but backward-compatible.

    **Important:** Some functions have complex try-catch structures or access getAuthorizedProject result's `project` field. When changing from `await getAuthorizedProject(id)` to `const auth = await getAuthorizedProject(id)`, ensure any subsequent use of `project` from the auth result is updated to `auth.project` if the code previously destructured it.

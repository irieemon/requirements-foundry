From lib/auth/authorization.ts (after Plan 01):
```typescript
export type ProjectRole = 'admin' | 'owner' | 'editor' | 'viewer';

export interface AuthResult {
  project: Project;
  user: UserInfo;
  role: ProjectRole;
  canEdit: boolean;
  isAdmin: boolean;
}

export interface AuthResultWithEntity<T> extends AuthResult {
  entity: T;  // The resolved child entity (Run, etc.)
}

export async function getAuthorizedProject(projectId: string): Promise<AuthResult>;
export async function getAuthorizedRun(runId: string): Promise<AuthResultWithEntity<Run>>;
// getAuthorizedRun calls notFound() if run not found or user has no access
// In API routes, notFound() throws — must be caught and converted to JSON 404
```

Current inline pattern in API routes (to be replaced):
```typescript
// BEFORE:
const user = await getCurrentUser();
const run = await db.run.findUnique({ where: { id }, include: { project: { select: { userId: true } } } });
if (!run || (run.project.userId !== user.email && !isAdmin(user.email))) {
  return NextResponse.json({ error: "Run not found" }, { status: 404 });
}

// AFTER:
let auth;
try {
  auth = await getAuthorizedRun(id);
} catch {
  return NextResponse.json({ error: "Run not found" }, { status: 404, headers: CACHE_HEADERS });
}
const { entity: run } = auth;
```

  <action>
    Create `components/bug-report/bug-report-button.tsx` as a `"use client"` component.

    **Props interface:**
    ```typescript
    interface BugReportButtonProps {
      user: UserInfo;
    }
    ```

    **State variables:**
    - `open: boolean` (dialog open state, default false)
    - `description: string` (textarea value, default "")
    - `submitting: boolean` (loading state, default false)
    - `cooldown: boolean` (30s post-submit cooldown, default false)
    - `touched: boolean` (whether textarea has been blurred, default false, for validation display)

    **Imports:**
    ```typescript
    "use client";

    import { useState } from "react";
    import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogTrigger, DialogFooter } from "@/components/ui/dialog";
    import { Button } from "@/components/ui/button";
    import { Textarea } from "@/components/ui/textarea";
    import { Tooltip, TooltipContent, TooltipTrigger } from "@/components/ui/tooltip";
    import { Label } from "@/components/ui/label";
    import { Bug, Loader2, MapPin } from "lucide-react";
    import { toast } from "sonner";
    import { submitBugReport } from "@/server/actions/bug-reports";
    import type { UserInfo } from "@/lib/auth/types";
    ```

    **handleSubmit function:**
    - Guard: if `description.trim().length < 10`, return early
    - Set `submitting = true`
    - Call `submitBugReport({ description: description.trim(), pageUrl: window.location.pathname, browserMetadata: JSON.stringify({ userAgent: navigator.userAgent, viewport: { width: window.innerWidth, height: window.innerHeight } }) })`
    - If `result.success`:
      - `toast.success("Bug report submitted", { description: "Thanks for helping us improve." })`
      - Reset: `setDescription("")`, `setOpen(false)`, `setTouched(false)`
      - Start cooldown: `setCooldown(true)`, `setTimeout(() => setCooldown(false), 30000)`
    - If `!result.success`:
      - `toast.error("Failed to submit report", { description: "Something went wrong. Please try again.", duration: 5000 })`
    - Finally: `setSubmitting(false)`

    **handleOpenChange function:**
    - `setOpen(isOpen)`
    - If closing (!isOpen): `setDescription("")`, `setTouched(false)`

    **Validation logic:**
    - `const isValid = description.trim().length >= 10`
    - `const showError = touched && description.trim().length > 0 && !isValid`

    **JSX structure:**

    1. **FAB (floating action button):** Wrapped in Tooltip. Only visible when `!open`.
       ```jsx
       {!open && (
         <Tooltip>
           <TooltipTrigger asChild>
             <DialogTrigger asChild>
               <button
                 className="fixed bottom-20 right-6 md:bottom-6 z-50 h-12 w-12 rounded-full bg-primary text-primary-foreground shadow-lg hover:bg-primary/90 focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 active:scale-95 transition-opacity duration-200 flex items-center justify-center disabled:opacity-50 disabled:cursor-not-allowed"
                 disabled={cooldown}
                 aria-label="Report a bug"
               >
                 <Bug className="h-5 w-5" />
               </button>
             </DialogTrigger>
           </TooltipTrigger>
           <TooltipContent side="left">
             {cooldown ? "Report submitted recently" : "Report Bug"}
           </TooltipContent>
         </Tooltip>
       )}
       ```

    2. **Dialog content:** `sm:max-w-md`
       ```jsx
       <DialogContent className="sm:max-w-md">
         <DialogHeader>
           <DialogTitle>Report a Bug</DialogTitle>
         </DialogHeader>
         <div className="space-y-4">
           <div className="space-y-2">
             <Label htmlFor="bug-description" className="text-sm font-semibold">What went wrong?</Label>
             <Textarea
               id="bug-description"
               placeholder="Describe the issue you encountered..."
               value={description}
               onChange={(e) => setDescription(e.target.value.slice(0, 2000))}
               onBlur={() => setTouched(true)}
               rows={4}
               className="resize-none"
               disabled={submitting}
             />
             <div className="flex justify-between items-center">
               {showError ? (
                 <p className="text-xs text-destructive" aria-live="polite">Please describe the issue in at least 10 characters.</p>
               ) : <span />}
               <p className="text-xs text-muted-foreground">{description.length}/2000</p>
             </div>
           </div>
           <div className="flex items-center gap-1 text-xs text-muted-foreground">
             <MapPin className="h-3 w-3" />
             <span>Page: {typeof window !== "undefined" ? window.location.pathname : ""}</span>
           </div>
         </div>
         <DialogFooter>
           <Button variant="outline" onClick={() => setOpen(false)} disabled={submitting}>Never mind</Button>
           <Button onClick={handleSubmit} disabled={!isValid || submitting || cooldown}>
             {submitting ? (
               <><Loader2 className="mr-2 h-4 w-4 animate-spin" />Submitting...</>
             ) : (
               "Submit Report"
             )}
           </Button>
         </DialogFooter>
       </DialogContent>
       ```

    3. **Wrap everything in Dialog:**
       ```jsx
       <Dialog open={open} onOpenChange={handleOpenChange}>
         {/* FAB trigger */}
         {/* DialogContent */}
       </Dialog>
       ```

    **NOTE on page URL hint:** The `window.location.pathname` in the page URL hint area is rendered inside JSX. Since this is a client component and the MapPin text is inside the dialog (which only renders when open, triggered by user click), `window` is always available. Do NOT use `typeof window !== "undefined"` guard inside the dialog body -- only use it if rendering outside the dialog. Actually, to be safe for SSR of the static part, use the guard as shown above.

    Export as named export: `export function BugReportButton({ user }: BugReportButtonProps)`
